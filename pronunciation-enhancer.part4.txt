  function consonantTongue(place,manner){
    const near = manner === "fricative" || manner === "approximant";
    const gap = near ? 2.1 : .2;
    switch(place){
      case "Dental":
        return `M15.5,43 C20,39 25,40 33,43 C46,44 57,49 59,60 C60,69 56,74 49,75 C37,77 25,69 20,59 C18,54 16,48 15.5,43 Z`;
      case "Alveolar":
        return `M20,49 C21,${(35+gap).toFixed(1)} 24,${(33.6+gap).toFixed(1)} 27,${(34+gap).toFixed(1)} C35,36 49,39 56,49 C61,58 61,69 54,74 C45,80 31,72 24,63 C20,58 19,53 20,49 Z`;
      case "Post-alveolar":
        return `M20,49 C23,41 28,${(33+gap).toFixed(1)} 35,${(31.7+gap).toFixed(1)} C43,31 54,39 58,50 C62,61 59,71 51,75 C40,79 27,69 22,60 C19,55 19,52 20,49 Z`;
      case "Velar":
        return `M20,49 C28,45 38,44 47,${(39+gap).toFixed(1)} C52,${(35+gap).toFixed(1)} 56,${(34+gap).toFixed(1)} 59,${(37+gap).toFixed(1)} C62,45 61,61 57,69 C53,77 41,78 31,70 C23,64 18,55 20,49 Z`;
      default:
        return `M20,49 C26,43 35,41 44,42 C53,43 58,49 59,59 C60,68 57,73 50,75 C39,78 26,69 21,59 C19,55 18,52 20,49 Z`;
    }
  }

  function enhanceConsonantDetail(sym){
    const info = consonants[sym];
    const svg = document.querySelector("#detail .anatomy-mini");
    if(!info || !svg) return;
    const [place,manner,voiced] = info;
    svg.dataset.place = place.toLowerCase().replace(/[^a-z]+/g,"-");
    svg.dataset.manner = manner;
    svg.dataset.voiced = String(voiced);
    svg.dataset.active = "1";
    // approximants (w/l/r/y) hold their position like a vowel — no continuous airflow/buzz
    svg.classList.toggle("an-speaking",!!voiced && manner !== "approximant");

    const tongue = svg.querySelector(".an-tongue");
    const velum = svg.querySelector(".an-velum");
    const uvula = svg.querySelector(".an-uvula");
    const upperLip = svg.querySelector(".an-upper-lip");
    const lowerLip = svg.querySelector(".an-lower-lip");
    if(tongue) tongue.setAttribute("d",consonantTongue(place,manner));
    if(place === "Bilabial"){
      upperLip?.setAttribute("d","M8,38 Q4,40 7,42 Q11,42 15,41.5");
      lowerLip?.setAttribute("d","M8,45 Q4,43.5 7,42.2 Q11,42.1 15,42.5");
    }else if(place === "Labio-dental"){
      lowerLip?.setAttribute("d","M8,46 Q7,42 12.4,39.8 Q14,40.5 16,42.5");
    }
    if(manner === "nasal"){
      velum?.setAttribute("d","M58,31 Q61,38 59,46");
      if(uvula){ uvula.setAttribute("cx","58.8"); uvula.setAttribute("cy","47"); }
    }

    const step = svg.closest(".step");
    if(!step || step.querySelector(".an-motion-note")) return;
    const mannerText = {
      plosive:"The articulators seal, pressure builds, then the air releases.",
      fricative:"The articulators leave a narrow channel so the airflow becomes noisy.",
      affricate:"The movement begins with a seal and opens into a narrow fricative channel.",
      nasal:"The velum lowers so the air is redirected through the nasal cavity.",
      approximant:"The articulators move close together without making a full seal."
    }[manner];
    const note = document.createElement("div");
    note.className = "an-motion-note";
    note.innerHTML = `<b>Animated cutaway:</b> ${mannerText}${voiced?" The vocal folds vibrate.":" The vocal folds stay apart."} <button class="btn" type="button">↻ Replay</button>`;
